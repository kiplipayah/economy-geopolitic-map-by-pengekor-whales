function submitNewPassword(email) { var newPass = document.getElementById('pass-new').value.trim(); if (!newPass) { alert("⚠️ Masukkan password baru!"); return; } var btn = document.getElementById('btn-update-pass'); var originalText = btn ? btn.innerText : ""; if (btn) { btn.innerText = "⏳ Memperbarui di Database Sheets..."; btn.disabled = true; } // Mengirim parameter di URL dan Body (Kebal Redirect 302 Browser) var requestUrl = apiUrl + "?action=updateProfile&email=" + encodeURIComponent(email) + "&newPassword=" + encodeURIComponent(newPass); fetch(requestUrl, { method: "POST", body: JSON.stringify({ action: "updateProfile", email: email, newPassword: newPass }) }) .then(r => r.json()) .then(res => { if (btn) { btn.innerText = originalText; btn.disabled = false; } // MEMASTIKAN RESPON BENAR-BENAR DARI HASIL UPDATE PASSWORD if (res && res.status === "success" && res.type === "updateProfile_response") { alert("✅ Password berhasil diperbarui di Google Sheets! Silakan coba login ulang."); closeModal(); } else if (res && res.status === "error" && res.type === "updateProfile_response") { alert("❌ Gagal: " + (res.message || "Terjadi kesalahan pada database.")); } else { alert("⚠️ Respon server tidak valid, silakan coba lagi."); } }) .catch(e => { if (btn) { btn.innerText = originalText; btn.disabled = false; } alert("❌ KONEKSI GAGAL! Keterangan: " + e.toString()); }); }
