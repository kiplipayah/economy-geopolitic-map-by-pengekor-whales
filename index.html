/* FORMATTER ZONA IKLIM (UBAH KODE MENTAH DATABASE JADI RAMAH PENGGUNA) */ function formatZonaIklim(code) { if (!code) return '-'; var c = code.toString().trim().toUpperCase(); if (c === 'NORTH_4SEASONS' || c === 'NORTH_4_SEASONS') return '🍂 4 Musim (Bumi Utara)'; if (c === 'SOUTH_4SEASONS' || c === 'SOUTH_4_SEASONS') return '🍁 4 Musim (Bumi Selatan)'; if (c === 'DESERT' || c === 'GURUN') return '🌵 Gurun / Arid (Kering Ekstrem)'; if (c === 'TROPICAL' || c === 'TROPIS') return '🌴 Tropis (Panas & Lembap)'; if (c === 'EQUATORIAL') return '☀️ Ekuatorial'; if (c === 'SUBTROPICAL' || c === 'SUBTROPIS') return '🌤️ Subtropis'; if (c === 'MEDITERRANEAN') return '🌊 Mediterania'; if (c === 'ARCTIC' || c === 'POLAR') return '❄️ Arktik / Kutub'; return code; // Fallback jika teks di database sudah berbentuk deskripsi biasa }
